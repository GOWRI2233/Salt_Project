// Sensor Pins
#define MQ135_PIN 34
#define MQ7_PIN   35
// MQ Parameters
float R0_MQ135 = 8.50;
float R0_MQ7 = 0.39;
float Vc = 5.0;
float RL = 10.0;
unsigned long sendDataPrevMillis = 0;
// MQ-7 Rs Calculation
float calculateRs_MQ7(int analogValue) {
  float voltage = (analogValue / 4095.0) * Vc;
  float Rs = RL * (Vc - voltage) / voltage;
  return Rs;
}
// MQ-7 PPM
float calculatePPM_MQ7(float Rs) {
  float ratio = Rs / R0_MQ7;
  float ppm = pow((ratio / 34.668), -1.512);
  return ppm;
}
// MQ-135 NH3, Benzene, CO2 Calculations
float getNH3PPM(float Rs) {
  return 1.5 * pow((Rs / R0_MQ135), 3.8);
}
float getBenzenePPM(float Rs) {
  return 0.4 * pow((Rs / R0_MQ135), -1.5);
}
float getCO2PPM(float Rs) {
  return 116.6020682 * pow((Rs / R0_MQ135), 2.3);
}
// Air Quality Classification
String classifyAirQuality(float co2_ppm) {
  if (co2_ppm <= 50) return "Good - No health implications";
  else if (co2_ppm <= 100) return "Moderate - Mild discomfort for sensitive individuals";
  else if (co2_ppm <= 150) return "Unhealthy for Sensitive Groups";
  else if (co2_ppm <= 200) return "Unhealthy - Health effects possible";
  else if (co2_ppm <= 300) return "Very Unhealthy - Emergency warnings";
  return "Hazardous - Serious health risk";
}
void setup() {
  Serial.begin(115200);
  Serial.println("Starting Air Quality & CO Monitoring...");
  // WiFi Connect
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to WiFi");
while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }
 Serial.println("\nWiFi Connected!");
// Firebase Setup
  config.database_url = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  auth.user.email = EMAIL;
  auth.user.password = PASS;

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}
void loop() {
  if (millis() - sendDataPrevMillis > 15000 || sendDataPrevMillis == 0) {
    sendDataPrevMillis = millis();
 // MQ135 Readings
    int mq135_raw = analogRead(MQ135_PIN);
    float Rs_MQ135 = ((1023.0 / mq135_raw) - 1) * R0_MQ135;
    float nh3_ppm = getNH3PPM(Rs_MQ135);
    float benzene_ppm = getBenzenePPM(Rs_MQ135);
    float co2_ppm = getCO2PPM(Rs_MQ135);
    String airQuality = classifyAirQuality(co2_ppm);
    // MQ7 Readings
    int mq7_raw = analogRead(MQ7_PIN);
    float Rs_MQ7 = calculateRs_MQ7(mq7_raw);
    float co_ppm = calculatePPM_MQ7(Rs_MQ7);
    // Print Results
    Serial.printf("NH3: %.2f ppm\n", nh3_ppm);
    Serial.printf("Benzene: %.2f ppm\n", benzene_ppm);
    Serial.printf("CO2: %.2f ppm\n", co2_ppm);
    Serial.println("Air Quality: " + airQuality);
    Serial.printf("CO: %.2f ppm\n\n", co_ppm);
    // Firebase JSON
    json.set("/NH3", nh3_ppm);
    json.set("/Benzene", benzene_ppm);
    json.set("/CO2", co2_ppm);
    json.set("/CO", co_ppm);
    json.set("/AirQuality", airQuality);
    // Send to Firebase
    if (Firebase.updateNode(firebaseData, "/airQualityData", json)) {
      Serial.println("Data sent to Firebase successfully.");
    } else {
      Serial.println("Failed to send data: " + firebaseData.errorReason());
    }
  }
}
